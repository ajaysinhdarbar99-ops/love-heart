    void charEl.offsetWidth;          // restart animation

    let k = 0;
    timer = setInterval(function () {
      k++;
      textEl.textContent = line.text.slice(0, k);
      if (k >= line.text.length) finishTyping();
    }, TYPE_MS);
  }

  function finishTyping() {
    clearInterval(timer);
    textEl.textContent = LINES[index].text;
    typed = true;
    setArrow(true);
  }

  // Tap anywhere: first finish the text, then go to the next line.
  document.getElementById("stage").addEventListener("pointerdown", function () {
    if (!typed) {
      finishTyping();
      return;
    }
    if (index < LINES.length - 1) {
      index++;
      showLine(index);
    }
  });

  showLine(0);
})();
